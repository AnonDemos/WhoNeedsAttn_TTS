    <script>
    (function () {
        var btn = document.getElementById('rxn-btn');
        var status = document.getElementById('rxn-status');
        var MODEL_MS = 68;
        var timer = null;
        var readyAt = 0;

        function armRound() {
            btn.dataset.state = 'armed';
            btn.textContent = 'Wait…';
            status.textContent = 'Wait for green…';
            status.classList.remove('result');
            var delay = 2000 + Math.random() * 2000;
            timer = setTimeout(function () {
                readyAt = performance.now();
                btn.dataset.state = 'ready';
                btn.textContent = 'Click!';
                status.textContent = 'Go!';
            }, delay);
        }

        btn.addEventListener('click', function () {
            var state = btn.dataset.state;
            if (state === 'idle' || state === 'done' || state === 'toosoon') {
                armRound();
                return;
            }
            if (state === 'armed') {
                clearTimeout(timer);
                btn.dataset.state = 'toosoon';
                btn.textContent = 'Too soon';
                status.textContent = 'You clicked before green — press the button to try again.';
                return;
            }
            if (state === 'ready') {
                var reaction = Math.round(performance.now() - readyAt);
                btn.dataset.state = 'done';
                btn.textContent = 'Play again';
                var pct = Math.abs(((reaction - MODEL_MS) / MODEL_MS) * 100).toFixed(1);
                var verb = reaction > MODEL_MS ? 'slower' : 'faster';
                status.innerHTML = 'You reacted in <b>' + reaction + ' ms</b> — ' + pct + '% ' + verb + " than the model's " + MODEL_MS + ' ms response time.';
                status.classList.add('result');
            }
        });
    })();
    </script>
